    <script>
        // DOM 元素
        const uploadArea = document.getElementById('uploadArea');
        const fileInput = document.getElementById('fileInput');
        const workspace = document.getElementById('workspace');
        const qualitySlider = document.getElementById('qualitySlider');
        const qualityText = document.getElementById('qualityText');
        const originalImg = document.getElementById('originalImg');
        const compressedImg = document.getElementById('compressedImg');
        const originalSize = document.getElementById('originalSize');
        const compressedSize = document.getElementById('compressedSize');
        const downloadBtn = document.getElementById('downloadBtn');
        const resetBtn = document.getElementById('resetBtn');

        // 全局变量
        let currentFile = null;
        let originalFileName = '';
        let compressedBlob = null;
        let outputMimeType = 'image/jpeg';

        // 格式化文件大小
        function formatSize(bytes) {
            if (bytes === 0) return '0 B';
            const k = 1024;
            const sizes = ['B', 'KB', 'MB'];
            const i = Math.floor(Math.log(bytes) / Math.log(k));
            return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
        }

        // 触发文件选择
        uploadArea.addEventListener('click', () => fileInput.click());

        // 拖拽事件处理
        uploadArea.addEventListener('dragover', (e) => {
            e.preventDefault();
            uploadArea.classList.add('dragover');
        });
        uploadArea.addEventListener('dragleave', () => {
            uploadArea.classList.remove('dragover');
        });
        uploadArea.addEventListener('drop', (e) => {
            e.preventDefault();
            uploadArea.classList.remove('dragover');
            if (e.dataTransfer.files.length) {
                handleFile(e.dataTransfer.files[0]);
            }
        });

        // 文件选择处理
        fileInput.addEventListener('change', (e) => {
            if (e.target.files.length) {
                handleFile(e.target.files[0]);
            }
        });

        // 处理文件
        function handleFile(file) {
            if (!file.type.match(/image\/(jpeg|png|webp)/)) {
                alert('请上传 JPG, PNG 或 WebP 格式的图片！');
                return;
            }

            currentFile = file;
            originalFileName = file.name;
            originalSize.textContent = formatSize(file.size);
            
            // 注：Canvas 的 toBlob 方法中，PNG 不支持调节压缩质量(quality)
            // 为了保证压缩效果并保留透明度，如果原图是 PNG，输出时默认转为 WebP
            outputMimeType = file.type;
            if (outputMimeType === 'image/png') {
                outputMimeType = 'image/webp'; 
            }

            const reader = new FileReader();
            reader.onload = (e) => {
                originalImg.src = e.target.result;
                originalImg.onload = () => {
                    uploadArea.style.display = 'none';
                    workspace.style.display = 'block';
                    compressImage(); // 初次加载时执行一次压缩
                };
            };
            reader.readAsDataURL(file);
        }

        // 核心压缩逻辑 (使用 Canvas API)
        function compressImage() {
            const quality = parseFloat(qualitySlider.value);
            qualityText.textContent = Math.round(quality * 100) + '%';

            const canvas = document.createElement('canvas');
            const ctx = canvas.getContext('2d');

            canvas.width = originalImg.naturalWidth;
            canvas.height = originalImg.naturalHeight;

            // 如果是转为 JPEG，填充白色背景防止透明区域变黑
            if (outputMimeType === 'image/jpeg') {
                ctx.fillStyle = '#fff';
                ctx.fillRect(0, 0, canvas.width, canvas.height);
            }

            ctx.drawImage(originalImg, 0, 0, canvas.width, canvas.height);

            // 执行压缩
            canvas.toBlob((blob) => {
                if (!blob) return;
                
                compressedBlob = blob;
                compressedSize.textContent = formatSize(blob.size);
                
                // 释放旧的 URL 避免内存泄漏
                if (compressedImg.src) {
                    URL.revokeObjectURL(compressedImg.src);
                }
                
                const newUrl = URL.createObjectURL(blob);
                compressedImg.src = newUrl;

            }, outputMimeType, quality);
        }

        // 监听滑块变化
        qualitySlider.addEventListener('input', compressImage);

        // 下载逻辑
        downloadBtn.addEventListener('click', () => {
            if (!compressedBlob) return;
            
            const link = document.createElement('a');
            // 生成新文件名
            const extension = outputMimeType.split('/')[1];
            const nameWithoutExt = originalFileName.substring(0, originalFileName.lastIndexOf('.')) || originalFileName;
            link.download = `${nameWithoutExt}_min.${extension}`;
            link.href = URL.createObjectURL(compressedBlob);
            link.click();
        });

        // 重新上传
        resetBtn.addEventListener('click', () => {
            workspace.style.display = 'none';
            uploadArea.style.display = 'block';
            fileInput.value = '';
            currentFile = null;
        });
    </script>
